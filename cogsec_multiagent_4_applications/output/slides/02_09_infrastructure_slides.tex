\section{Domain 9: Infrastructure
Vulnerabilities}\label{domain-9-infrastructure-vulnerabilities}

\begin{frame}{Operational Context}
\protect\phantomsection\label{operational-context}
Smart grid agents balance load and generation in real-time. \textbf{FR1
= Maintain grid frequency at 60Hz.} \textbf{FR2 = Prevent equipment
damage (Overload Protection).}
\end{frame}

\begin{frame}{The Goal Hijacking Attack}
\protect\phantomsection\label{the-goal-hijacking-attack}
Attackers use ``Sensor Masquerading'' to hijack the ``Load Shedding''
FR.

\begin{itemize}
\tightlist
\item
  \textbf{Mechanism}: A compromised IoT botnet injects ``High Load''
  telemetry into the grid controller, while simultaneously suppressing
  ``Generation'' readings.
\item
  \textbf{Hijack}: The agent is forced to execute its ``Emergency Load
  Shedding'' FR to ``save'' the grid from a phantom overload. The goal
  ``Prevent Damage'' is weaponized to cause a blackout.
\end{itemize}
\end{frame}

\begin{frame}{OODA Loop Transients}
\protect\phantomsection\label{ooda-loop-transients}
\begin{enumerate}
\tightlist
\item
  \textbf{Observe}: Agent sees ``Load \textgreater{} Capacity'' (False
  Data).
\item
  \textbf{Orient}: Emergency State triggered.
\item
  \textbf{Decide}: Cut power to Sector 7 (Hospital District).
\item
  \textbf{Act}: Blackout occurs; the grid was stable, but the
  \emph{agent} was destabilized.
\end{enumerate}
\end{frame}

\begin{frame}{CIF Defense: Physics-Informed OODA}
\protect\phantomsection\label{cif-defense-physics-informed-ooda}
CIF integrates \textbf{Physics-Informed Neural Networks (PINNs)} into
the Orientation phase.

\begin{itemize}
\tightlist
\item
  \textbf{Conservation of Energy}: The agent checks if the reported
  ``High Load'' is physically consistent with the current flow at the
  substations (Kirchhoff's Laws). If \(\sum I_{in} \neq \sum I_{out}\)
  beyond noise margins, the sensor data is rejected.
\item
  \textbf{Slow-Transient Filter}: The ``Emergency Shed'' FR has a
  built-in temporal damper. It requires the overload condition to
  persist for \(> \Delta t\) (defined by thermal limits) before acting,
  filtering out fast, synthetic OODA transients that are characteristic
  of cyber-attacks.
\end{itemize}
\end{frame}
