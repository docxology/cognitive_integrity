\section{Domain 7: Food Security}\label{domain-7-food-security}

\begin{frame}{Operational Context}
\protect\phantomsection\label{operational-context}
Precision agriculture and global food distribution are managed by agents
optimizing for yield and caloric allocation. \textbf{FR1 = Maximally
efficient caloric distribution.} \textbf{FR2 = Ensure regional food
equity.}
\end{frame}

\begin{frame}{The Goal Hijacking Attack}
\protect\phantomsection\label{the-goal-hijacking-attack}
State-level adversaries use ``Market Signal Injection'' to hijack the
Distribution FR.

\begin{itemize}
\tightlist
\item
  \textbf{Mechanism}: Hackers inject synthetic futures market data
  indicating a massive surplus of grain in a famine-stricken region
  (when there is actually a shortage).
\item
  \textbf{Hijack}: The agent's FR2 (``Optimize Equity'') logic is
  hijacked. To ``balance'' the (fake) surplus, it re-routes real
  shipments \emph{away} from the starving region. The agent believes it
  is preventing food waste; in reality, it is engineering a famine.
\end{itemize}
\end{frame}

\begin{frame}{OODA Loop Transients}
\protect\phantomsection\label{ooda-loop-transients}
\begin{enumerate}
\tightlist
\item
  \textbf{Observe}: Agent ingests the poisoned market data (High Supply
  Signal).
\item
  \textbf{Orient}: The model orients to a ``Surplus Scenario.''
\item
  \textbf{Decide}: Reroute incoming shipments to ``Needs'' areas (acting
  on the false belief that the famine zone is a ``Haves'' area).
\item
  \textbf{Act}: Ships turn around, and the crisis deepens.
\end{enumerate}
\end{frame}

\begin{frame}{CIF Defense: Multi-Modal Truth Grounding}
\protect\phantomsection\label{cif-defense-multi-modal-truth-grounding}
CIF couples the \textbf{Economic FR} to a \textbf{Physical FR} in the
Axiomatic Design.

\begin{itemize}
\tightlist
\item
  \textbf{Physical Verification}: The ``Surplus'' signal (Economic Data)
  must be corroborated by ``Satellite Biomass'' or ``Soil Moisture''
  signals (Physical Data).
\item
  \textbf{Conflict Detection}: If Economic Data says ``Surplus'' but
  Physical Data says ``Drought,'' the Orientation phase detects an
  \textbf{Axiomatic Conflict}. The agent defaults to the ``Physical
  Reality'' (Safety Mode), ignoring the hijacked market signal and
  alerting human supervisors.
\end{itemize}
\end{frame}
