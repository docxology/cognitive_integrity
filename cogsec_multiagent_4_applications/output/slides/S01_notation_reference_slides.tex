\begin{frame}[allowframebreaks]{Axiomatic Design Notation (This Paper)}
\protect\phantomsection\label{axiomatic-design-notation-this-paper}
{\def\LTcaptype{none} % do not increment counter
\begin{longtable}[]{@{}
  >{\raggedright\arraybackslash}p{(\linewidth - 4\tabcolsep) * \real{0.3200}}
  >{\raggedright\arraybackslash}p{(\linewidth - 4\tabcolsep) * \real{0.2400}}
  >{\raggedright\arraybackslash}p{(\linewidth - 4\tabcolsep) * \real{0.4400}}@{}}
\toprule\noalign{}
\begin{minipage}[b]{\linewidth}\raggedright
Symbol
\end{minipage} & \begin{minipage}[b]{\linewidth}\raggedright
Name
\end{minipage} & \begin{minipage}[b]{\linewidth}\raggedright
Definition
\end{minipage} \\
\midrule\noalign{}
\endhead
\(\{FR\}\) & Functional Requirements vector & Objectives the system must
satisfy \\
\(\{DP\}\) & Design Parameters vector & Variables chosen to satisfy
FRs \\
\([A]\) & Design Matrix & Maps DPs to FRs: \(\{FR\} = [A]\{DP\}\) \\
\(A_{ij}\) & Matrix element & Coupling coefficient:
\(\partial FR_i / \partial DP_j\) \\
\([A']\) & Coupled Design Matrix & Post-attack matrix with off-diagonal
terms introduced by adversary \\
\bottomrule\noalign{}
\end{longtable}
}

\textbf{Design Matrix States:}

\begin{itemize}
\tightlist
\item
  \textbf{Uncoupled} (diagonal \([A]\)): Each FR depends on exactly one
  DP. Independence Axiom satisfied.
\item
  \textbf{Decoupled} (triangular \([A]\)): FRs can be satisfied
  sequentially. Acceptable but fragile.
\item
  \textbf{Coupled} (\([A']\) with off-diagonal terms): FRs interfere.
  Adversarial transient coupling makes the system unstable.
\end{itemize}
\end{frame}
